from PIL import Image, ImageDraw, ImageFont import os, re, base64, io, zipfile, html as htmlmod from bs4 import BeautifulSoup src = "/mnt/data/MeowGames_FINAL_FIXED.html" out_dir = "/mnt/data/meow_thumbnails" os.makedirs(out_dir, exist_ok=True) # Read the existing game list from the user's HTML. with open(src, "r", encoding="utf-8") as f: source_html = f.read() soup = BeautifulSoup(source_html, "html.parser") games = [(a.get("data-name","").strip(), a.get("data-category","Arcade").strip()) for a in soup.select("article.card")] # Fonts font_paths = [ "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf", ] bold_font_path = next(p for p in font_paths if os.path.exists(p)) regular_font_path = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf" # Category-specific visual palettes and simple original symbols. palettes = { "Puzzle": ((37, 20, 73), (112, 64, 190), "✦"), "Sports": ((8, 55, 73), (25, 145, 185), "●"), "Racing": ((72, 18, 28), (210, 65, 65), "➤"), "Arcade": ((17, 35, 72), (35, 120, 220), "◆"), "Adventure": ((22, 70, 48), (45, 165, 95), "★"), "Board": ((54, 38, 22), (170, 105, 45), "♟"), "Strategy": ((52, 28, 70), (160, 75, 190), "✚"), } def safe_filename(name): return re.sub(r"[^a-z0-9]+", "_", name.lower()).strip("_") + ".png" def make_thumbnail(name, category, idx): W, H = 640, 360 c1, c2, symbol = palettes.get(category, palettes["Arcade"]) img = Image.new("RGB", (W, H), c1) d = ImageDraw.Draw(img) # Gradient-like horizontal bands for x in range(W): t = x / (W - 1) col = tuple(int(c1[i]*(1-t) + c2[i]*t) for i in range(3)) d.line([(x, 0), (x, H)], fill=col) # Decorative gaming shapes for k in range(9): x = (idx * 73 + k * 91) % W y = (idx * 47 + k * 61) % H r = 12 + ((idx + k) % 5) * 7 d.ellipse((x-r, y-r, x+r, y+r), outline=(255,255,255), width=2) # Central abstract game emblem cx, cy = W//2, H//2 - 18 d.rounded_rectangle((cx-82, cy-62, cx+82, cy+62), radius=24, fill=(10, 14, 25), outline=(255,255,255), width=3) f_symbol = ImageFont.truetype(bold_font_path, 62) bbox = d.textbbox((0,0), symbol, font=f_symbol) d.text((cx-(bbox[2]-bbox[0])/2, cy-(bbox[3]-bbox[1])/2-6), symbol, font=f_symbol, fill=(255,255,255)) # Title title = name.title() # Keep title within width. f_title = ImageFont.truetype(bold_font_path, 36 if len(title) <= 17 else 28) bbox = d.textbbox((0,0), title, font=f_title) tw = bbox[2]-bbox[0] if tw > 590: f_title = ImageFont.truetype(bold_font_path, 24) bbox = d.textbbox((0,0), title, font=f_title) tw = bbox[2]-bbox[0] d.rounded_rectangle((24, 276, W-24, 338), radius=14, fill=(0,0,0)) d.text(((W-tw)/2, 288), title, font=f_title, fill=(255,255,255)) # Category badge f_cat = ImageFont.truetype(regular_font_path, 20) cb = d.textbbox((0,0), category, font=f_cat) cw = cb[2]-cb[0] d.rounded_rectangle((18, 18, 18+cw+28, 55), radius=12, fill=(0,0,0)) d.text((32, 25), category, font=f_cat, fill=(235,235,255)) path = os.path.join(out_dir, safe_filename(name)) img.save(path, optimize=True) return path paths = [make_thumbnail(name, cat, i) for i, (name, cat) in enumerate(games)] # Build a single self-contained HTML file: thumbnails are embedded, so the user # does not need to upload a separate images folder. for article in soup.select("article.card"): name = article.get("data-name","").strip() path = os.path.join(out_dir, safe_filename(name)) if os.path.exists(path): with open(path, "rb") as f: b64 = base64.b64encode(f.read()).decode("ascii") icon = article.select_one(".icon") if icon: icon.name = "img" icon.attrs = { "class": "game-thumb", "src": "data:image/png;base64," + b64, "alt": name.title(), "loading": "lazy" } # Replace the icon CSS with thumbnail styling. style = soup.find("style") if style: css = style.string or "" css = re.sub(r"\.icon\{[^}]*\}", "", css) css += """ .game-thumb{width:100%;aspect-ratio:16/9;object-fit:cover;border-radius:12px; display:block;margin-bottom:12px;border:1px solid #343a4a} .card{overflow:hidden} """ style.string = css final_html = os.path.join("/mnt/data", "MeowGames_FINAL_WITH_THUMBNAILS.html") with open(final_html, "w", encoding="utf-8") as f: f.write(str(soup)) # Also provide all thumbnails in a ZIP in case the user wants individual files. zip_path = "/mnt/data/MeowGames_91_Thumbnails.zip" with zipfile.ZipFile(zip_path, "w", zipfile.ZIP_DEFLATED) as z: for p in paths: z.write(p, arcname=os.path.basename(p)) print(f"Done: {len(games)} different game thumbnails created.") print(final_html) print(zip_path)
